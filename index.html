from datetime import date from html import escape from pathlib import Path from string import Template # Voeg nieuwe leden hier toe. Gebruik JJJJ-MM-DD voor de datum. LEDEN = [ ("Javier", "2012-11-15"), ("Eugene", "2012-11-15"), ("Vincent", "2012-11-15"), ("Khalid", "2013-07-15"), ("Ruud", "2015-05-03"), ("Sander", "2015-08-11"), ("Marc", "2017-11-02"), ("Sennen", "2022-12-15"), ("Jef", "2024-12-15"), ("Ron", "2026-05-15"), ] MAANDEN = ( "januari", "februari", "maart", "april", "mei", "juni", "juli", "augustus", "september", "oktober", "november", "december", ) PAGINA = Template(r''' De 15de | Ledenoverzicht
15

De 15de

Elke maand samen aan tafel

Hoe lang zijn we al lid?

$rows
Leden op volgorde van toetreding met een horizontaal staafdiagram van hun lidmaatschapsduur.
LidLid sindsDagen

Peildatum $reference_date

''') def maak_html(peildatum: date) -> str: """Bereken de duur en bouw de HTML voor de opgegeven peildatum.""" leden = sorted(LEDEN, key=lambda lid: lid[1]) rijen = [] for naam, datum in leden: lid_sinds = date.fromisoformat(datum) if lid_sinds > peildatum: continue dagen = (peildatum - lid_sinds).days dagen_tekst = f"{dagen:,}".replace(",", ".") rijen.append( f'{escape(naam)}' f'{lid_sinds:%d-%m-%Y}' f'{dagen_tekst}' ) datumtekst = f"{peildatum.day} {MAANDEN[peildatum.month - 1]} {peildatum.year}" return PAGINA.substitute(rows="\n".join(rijen), reference_date=datumtekst) if __name__ == "__main__": uitvoer = Path(__file__).resolve().with_suffix(".html") uitvoer.write_text(maak_html(date.today()), encoding="utf-8") print(f"HTML gemaakt: {uitvoer}")
